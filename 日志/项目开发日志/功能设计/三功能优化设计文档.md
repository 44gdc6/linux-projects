## 1. 硬件看门狗 (IWDG)

### 1.1 问题分析

当前 F407 和 F103 的主循环：

```c
// F407 main.c
while (1) {
    Debug_Heartbeat_Process();
    CAN_Link_Process();
    HAL_Delay(10);
}

// F103 main.c
while (1) {
    CAN_Link_Process();
    HAL_Delay(10);
}
```

**风险场景：**
- CAN 中断处理中死锁（如 `HAL_CAN_AddTxMessage` 永久等待）
- HardFault 未进入 Error_Handler（如总线错误触发 BusFault）
- 硬件异常导致程序跑飞但未触发异常向量

**后果：** 节点永久失联，i.MX6ULL 只能通过超时检测到离线，无法自动恢复。

### 1.2 IWDG 原理

```
LSI (32kHz/40kHz) → IWDG 预分频器 → 重装载寄存器 → 计数器
                                                     ↓
                                            计数到 0 → 系统复位
                                                     ↑
                                            喂狗 (Refresh) ← 主循环定期调用
```

- **时钟源：** LSI（低速内部振荡器），独立于系统时钟，即使 HSI/HSE 故障仍能工作
- **无中断：** IWDG 不产生中断，超时直接复位
- **寄存器写保护：** 需要先写 `0x5555` 到 `KR` 寄存器才能修改预分频和重装载值

### 1.3 超时参数计算

#### F407（LSI = 32 kHz）

| 预分频 | 重装载值 | 超时时间 | 适用场景 |
|--------|---------|---------|---------|
| /64 | 1250 | 2.5 秒 | 严格守护，快速复位 |
| /128 | 1250 | 5.0 秒 | 推荐：覆盖主循环 1s 周期 |
| /256 | 4000 | 32 秒 | 宽松守护，兼容慢速操作 |

**推荐配置：** 预分频 /128，重装载 1250，超时 5 秒。主循环每秒喂一次狗，留 5 倍余量。

#### F103（LSI = 40 kHz）

| 预分频 | 重装载值 | 超时时间 | 适用场景 |
|--------|---------|---------|---------|
| /64 | 1250 | 2.0 秒 | 严格守护 |
| /128 | 2500 | 8.0 秒 | 推荐：覆盖 CAN 处理周期 |
| /256 | 5000 | 32 秒 | 宽松守护 |

**推荐配置：** 预分频 /128，重装载 2500，超时 8 秒。

```c
void Error_Handler(void)
{
    /* 不喂狗 → IWDG 超时 → 自动复位 */
    while (1) {
        Boot_Log("ERROR: handler entered\r\n");
        HAL_Delay(500);
        /* 故意不喂狗，让看门狗复位系统 */
    }
}
```

### 1.5 注意事项

| 问题 | 说明 |
|------|------|
| IWDG 初始化时机 | 必须在所有其他外设初始化之后，否则初始化过程中的 HAL_Delay 可能触发复位 |
| 调试模式 | 调试器暂停 CPU 时 IWDG 仍在计数，会导致反复复位。需在调试时关闭 IWDG 或启用 `DBG_IWDG_STOP` |
| 低功耗 | Sleep/Stop 模式下 IWDG 继续运行（LSI 不关），可作为低功耗唤醒后的安全机制 |
| 喂狗位置 | 只在主循环喂狗，中断中不喂狗。如果主循环卡死，看门狗一定能复位 |

#### sensor_gps.c 核心逻辑

```c
int gps_init(const char *uart_path)
{
    int fd;
    struct termios tty;

    fd = open(uart_path, O_RDONLY | O_NOCTTY | O_NONBLOCK);
    if (fd < 0) return -1;

    /* 配置 9600 8N1 */
    tcgetattr(fd, &tty);
    cfsetispeed(&tty, B9600);
    cfsetospeed(&tty, B9600);
    tty.c_cflag &= ~PARENB;
    tty.c_cflag &= ~CSTOPB;
    tty.c_cflag &= ~CSIZE;
    tty.c_cflag |= CS8;
    tty.c_cflag &= ~CRTSCTS;
    tty.c_lflag &= ~(ICANON | ECHO | ECHOE | ISIG);
    tty.c_iflag &= ~(IXON | IXOFF | IXANY);
    tty.c_cc[VMIN] = 0;
    tty.c_cc[VTIME] = 10;  /* 100ms 超时 */
    tcsetattr(fd, TCSANOW, &tty);

    return fd;
}

int gps_read(int fd, gps_info_t *info)
{
    char buf[512];
    ssize_t n;
    fd_set rfds;
    struct timeval tv;

    FD_ZERO(&rfds);
    FD_SET(fd, &rfds);
    tv.tv_sec = 1;
    tv.tv_usec = 0;

    if (select(fd + 1, &rfds, NULL, NULL, &tv) <= 0) return -1;

    n = read(fd, buf, sizeof(buf) - 1);
    if (n <= 0) return -1;
    buf[n] = '\0';

    /* 解析 $GPRMC */
    char *rmc = strstr(buf, "$GPRMC");
    if (rmc == NULL) rmc = strstr(buf, "$GNRMC");
    if (rmc) parse_rmc(rmc, info);

    /* 解析 $GPGGA (补充卫星数/海拔) */
    char *gga = strstr(buf, "$GPGGA");
    if (gga == NULL) gga = strstr(buf, "$GNGGA");
    if (gga) parse_gga(gga, info);

    return 0;
}
```

#### NMEA 解析（从现有 gps.c 移植）

```c
static void parse_rmc(const char *sentence, gps_info_t *info)
{
    /* $GPRMC,hhmmss.ss,A,llll.ll,a,yyyyy.yy,a,x.x,x.x,ddmmyy,x.x,a,a*hh
     *
     * 字段: 1=时间, 2=状态(A=有效/V=无效), 3=纬度, 4=N/S,
     *        5=经度, 6=E/W, 7=速度(节), 8=航向, 9=日期
     */
    const char *p = sentence;
    int field = 0;
    char lat_str[16] = {0}, lon_str[16] = {0};
    char lat_dir = 'N', lon_dir = 'E';
    float speed_knots = 0;

    /* 跳过 tag */
    p = strchr(p, ',');
    if (p == NULL) return;

    while (*p && field < 9) {
        p++;
        field++;
        const char *start = p;
        while (*p && *p != ',' && *p != '*') p++;
        int len = (int)(p - start);

        switch (field) {
            case 2: info->valid = (*start == 'A') ? 1 : 0; break;
            case 3: strncpy(lat_str, start, len < 15 ? len : 15); break;
            case 4: lat_dir = *start; break;
            case 5: strncpy(lon_str, start, len < 15 ? len : 15); break;
            case 6: lon_dir = *start; break;
            case 7: speed_knots = atof(start); break;
            case 8: info->course = (float)atof(start); break;
        }
    }

    if (info->valid) {
        info->latitude = convert_to_degree(lat_str);
        if (lat_dir == 'S') info->latitude = -info->latitude;
        info->longitude = convert_to_degree(lon_str);
        if (lon_dir == 'W') info->longitude = -info->longitude;
        info->speed = speed_knots * 1.852f;  /* 节 → km/h */
    }
}

static double convert_to_degree(const char *nmea_coord)
{
    /* NMEA 格式: ddmm.mmmm 或 dddmm.mmmm */
    double raw = atof(nmea_coord);
    int deg = (int)(raw / 100);
    double min = raw - deg * 100.0;
    return deg + min / 60.0;
}
```

### 2.5 电子围栏设计

#### 围栏数据结构

```c
typedef struct {
    double center_lat;      /* 围栏中心纬度 */
    double center_lon;      /* 围栏中心经度 */
    double radius_meters;   /* 围栏半径 (米) */
    uint8_t enabled;        /* 1=启用围栏检测 */
} geofence_config_t;
```

#### Haversine 公式计算两点距离

```c
#define EARTH_RADIUS_METERS 6371000.0

double geofence_distance_meters(double lat1, double lon1, double lat2, double lon2)
{
    double dlat = (lat2 - lat1) * M_PI / 180.0;
    double dlon = (lon2 - lon1) * M_PI / 180.0;
    double a = sin(dlat/2) * sin(dlat/2) +
               cos(lat1 * M_PI / 180.0) * cos(lat2 * M_PI / 180.0) *
               sin(dlon/2) * sin(dlon/2);
    double c = 2 * atan2(sqrt(a), sqrt(1-a));
    return EARTH_RADIUS_METERS * c;
}

int geofence_check(const geofence_config_t *fence,
                   double lat, double lon)
{
    if (!fence->enabled) return 0;  /* 围栏未启用 */

    double dist = geofence_distance_meters(
        fence->center_lat, fence->center_lon, lat, lon);

    return (dist > fence->radius_meters) ? 1 : 0;  /* 1=越界 */
}
```

#define GEOFENCE_CENTER_LON     108.9404

#### gps_thread 集成围栏

```c
void *gps_thread(void *arg)
{
    gps_info_t gps;
    geofence_config_t fence = {
        .center_lat = GEOFENCE_CENTER_LAT,
        .center_lon = GEOFENCE_CENTER_LON,
        .radius_meters = GEOFENCE_RADIUS_METERS,
        .enabled = GEOFENCE_ENABLED
    };
    int prev_alarm = 0;

    (void)arg;
    int fd = gps_init("/dev/ttymxc2");
    if (fd < 0) {
        LOG_ERROR("gps_thread: uart init failed");
        return NULL;
    }

    LOG_INFO("gps_thread started on /dev/ttymxc2");

    while (1) {
        data_t sample;
        memset(&sample, 0, sizeof(sample));
        sample.sample_time = time(NULL);

        if (gps_read(fd, &gps) == 0 && gps.valid) {
            sample.gps_latitude = gps.latitude;
            sample.gps_longitude = gps.longitude;
            sample.gps_altitude = gps.altitude;
            sample.gps_speed = gps.speed;
            sample.gps_course = gps.course;
            sample.gps_satellites = gps.satellites;
            sample.gps_valid = 1;

            /* 电子围栏检测 */
            sample.gps_fence_alarm = geofence_check(
                &fence, gps.latitude, gps.longitude);

            if (sample.gps_fence_alarm && !prev_alarm) {
                LOG_WARN("GEO FENCE ALARM: vehicle outside safe zone! "
                         "lat=%.6f lon=%.6f",
                         gps.latitude, gps.longitude);
            }
            prev_alarm = sample.gps_fence_alarm;
        }

        mailbox_send_msg("mqtt_thread", sample);
        mailbox_send_msg("lvgl_thread", sample);
        mailbox_send_msg("sqlite_thread", sample);

        sleep(1);  /* 每秒更新一次 */
    }

    gps_deinit(fd);
    return NULL;
}
```

## 3. 短信/推送报警

### 3.1 方案对比

| 方案 | 优点 | 缺点 | 推荐度 |
|------|------|------|--------|
| **A. SIM800L GSM 模块** | 独立于网络，短信直接到达手机 | 需额外硬件、SIM 卡、月费 | ⭐⭐⭐⭐ |
| **B. OneNET 平台告警规则** | 零硬件改动，服务器端配置 | 依赖平台，延迟不确定 | ⭐⭐⭐ |
| **C. MQTT → 自建服务器 → WebSocket/App 推送** | 灵活可控 | 需要服务器，复杂度高 | ⭐⭐ |

**推荐方案 A (SIM800L)：** 与项目"嵌入式硬件"定位一致，可控性最强，适合毕设/产品演示。

### 3.2 SIM800L 硬件连接

```
SIM800L 模块
  VCC  → 5V (外部供电，峰值 2A)
  GND  → GND
  TXD  → i.MX6ULL UART1_RX (UART1_IO0 / GPIO_03)
  RXD  → i.MX6ULL UART1_TX (UART1_IO1 / GPIO_04)
  RST  → GPIO 控制 (可选，用于模块重启)
```

**注意：** SIM800L 供电需要独立 5V/2A 电源，不能直接从 i.MX6ULL 取电，否则电压跌落导致重启。

| 酒精超标 | `alcohol_level == HIGH` 连续 3 次 | 🔴 紧急 | "【紧急】驾驶舱酒精浓度超标！" |
| 电子围栏越界 | `gps_fence_alarm == 1` | 🟡 警告 | "【警告】车辆驶出安全区域！当前位置: lat, lon" |
| 温度异常 | `temp > 50°C` 或 `temp < -10°C` | 🟡 警告 | "【警告】货舱温度异常: xx°C" |
| 节点离线 | CAN 节点超时 30 秒 | 🟡 警告 | "【警告】F407/F103 节点离线" |

### 3.5 防重复报警机制

```c
typedef struct {
    uint32_t last_alarm_time[ALARM_TYPE_COUNT];  /* 上次报警时间戳 */
    uint32_t cooldown_sec;                        /* 冷却时间 (秒) */
} alarm_rate_limiter_t;

int alarm_should_send(alarm_rate_limiter_t *limiter, alarm_type_t type)
{
    uint32_t now = (uint32_t)time(NULL);
    if ((now - limiter->last_alarm_time[type]) < limiter->cooldown_sec) {
        return 0;  /* 冷却中，不发送 */
    }
    limiter->last_alarm_time[type] = now;
    return 1;
}
```

**冷却时间：** 60 秒（同一类型报警 1 分钟内只发一次短信）。

#### sms.c 核心逻辑

```c
static int sms_send_at(int fd, const char *cmd, const char *expect, int timeout_ms)
{
    char buf[256];
    fd_set rfds;
    struct timeval tv;

    /* 发送 AT 命令 */
    write(fd, cmd, strlen(cmd));
    write(fd, "\r\n", 2);

    /* 等待响应 */
    FD_ZERO(&rfds);
    FD_SET(fd, &rfds);
    tv.tv_sec = timeout_ms / 1000;
    tv.tv_usec = (timeout_ms % 1000) * 1000;

    if (select(fd + 1, &rfds, NULL, NULL, &tv) <= 0) return -1;

    int n = read(fd, buf, sizeof(buf) - 1);
    if (n <= 0) return -1;
    buf[n] = '\0';

    return strstr(buf, expect) ? 0 : -1;
}

int sms_send(const char *phone, const char *message)
{
    char cmd[128];
    char ctrl_z = 0x1A;

    /* 发送 AT+CMGS="phone" */
    snprintf(cmd, sizeof(cmd), "AT+CMGS=\"%s\"", phone);
    write(g_sms_fd, cmd, strlen(cmd));
    write(g_sms_fd, "\r\n", 2);
    usleep(500000);  /* 等待 > 提示符 */

    /* 发送短信内容 */
    write(g_sms_fd, message, strlen(message));
    write(g_sms_fd, &ctrl_z, 1);  /* Ctrl+Z 发送 */

    /* 等待 +CMGS 确认 */
    return sms_send_at_wait(g_sms_fd, "+CMGS", 10000);
}

void *sms_thread(void *arg)
{
    data_t sample;
    alarm_rate_limiter_t limiter = { .cooldown_sec = 60 };

    (void)arg;

    if (sms_init("/dev/ttymxc1") != 0) {
        LOG_ERROR("sms_thread: SIM800L init failed");
        return NULL;
    }

    LOG_INFO("sms_thread started");

    while (1) {
        if (mailbox_recv_msg(&sample) <= 0) {
            usleep(200000);
            continue;
        }

        /* 火焰报警 */
        if (sample.flame_valid && sample.flame_status == 1) {
            if (alarm_should_send(&limiter, ALARM_FIRE)) {
                char msg[256];
                snprintf(msg, sizeof(msg),
                    "ALARM:Fire detected! Lat=%.4f Lon=%.4f",
                    sample.gps_latitude, sample.gps_longitude);
                sms_send(ALARM_PHONE, msg);
                LOG_WARN("SMS sent: fire alarm");
            }
        }

        /* 酒精报警 */
        if (sample.alcohol_valid && sample.alcohol_level == ALCOHOL_LEVEL_HIGH) {
            if (alarm_should_send(&limiter, ALARM_ALCOHOL)) {
                sms_send(ALARM_PHONE,
                    "ALARM:Alcohol level HIGH in cockpit!");
                LOG_WARN("SMS sent: alcohol alarm");
            }
        }

        /* 围栏越界报警 */
        if (sample.gps_fence_alarm) {
            if (alarm_should_send(&limiter, ALARM_FENCE)) {
                char msg[256];
                snprintf(msg, sizeof(msg),
                    "ALARM:Vehicle outside geofence! Lat=%.4f Lon=%.4f",
                    sample.gps_latitude, sample.gps_longitude);
                sms_send(ALARM_PHONE, msg);
                LOG_WARN("SMS sent: geofence alarm");
            }
        }

        /* 温度异常报警 */
        if (sample.temp_valid && (sample.temp > 50.0 || sample.temp < -10.0)) {
            if (alarm_should_send(&limiter, ALARM_TEMP)) {
                char msg[128];
                snprintf(msg, sizeof(msg),
                    "ALARM:Cargo temp abnormal: %.1f C", sample.temp);
                sms_send(ALARM_PHONE, msg);
                LOG_WARN("SMS sent: temperature alarm");
            }
        }
    }

    sms_deinit();
    return NULL;
}
```

## 6. 风险与注意事项

| 风险 | 影响 | 缓解措施 |
|------|------|---------|
| IWDG 调试时反复复位 | 无法使用调试器 | 开发阶段在 hal_conf.h 中条件编译关闭 IWDG |
| SIM800L 供电不足 | 搜网/发短信时掉电 | 使用独立 5V/2A 电源，加 1000μF 电容 |
| GPS 冷启动定位慢 | 首次定位需 30-60 秒 | UI 显示"定位中..."，数据标记为 invalid |
| 短信延迟 | 报警不及时 | 同时通过 MQTT 推送，短信作为备份通道 |
| 电子围栏半径设置不当 | 频繁误报 | 围栏半径 ≥ 10km，冷却时间 60 秒 |
| AT 命令超时 | 短信线程卡死 | 每个 AT 命令设 5 秒超时，失败后重试 3 次 |
